```rust
/// Modo de autenticacion declarado en un endpoint (DSL v0.5).
#[derive(Debug, Clone, PartialEq, Eq, Default)]
pub enum AuthMode {
    /// Sin autenticacion requerida (valor por defecto).
    #[default]
    None,
    /// El token JWT es requerido; la Shield rechaza requests sin token.
    Required,
    /// El token JWT es opcional; el handler decide como tratar la ausencia.
    Optional,
}

impl AuthMode {
    /// Representacion textual del modo tal como aparece en el schema.
    pub fn as_str(&self) -> &'static str {
        match self {
            AuthMode::None => "none",
            AuthMode::Required => "required",
            AuthMode::Optional => "optional",
        }
    }
}

/// Declaracion de evento en DSL v0.6: `event nombre { payload T retain Nd }`.
#[derive(Debug, Clone)]
pub struct EventDef {
    /// Nombre del evento en formato `entidad.accion`.
    pub name: Spanned<std::string::String>,
    /// Nombre del tipo de payload (referencia a model/request/response).
    pub payload: Spanned<std::string::String>,
    /// Retencion en dias. `None` si no se declara.
    pub retain_days: Option<u32>,
    /// Span del bloque completo.
    pub span: Span,
}
```

```rust
// Rama para policy
Token::Policy => {
    if let Some((Token::StringLit(expr), sp)) = tokens.next() {
        endpoint_policy = Some(Spanned::new(expr, sp));
    } else {
        diags.push(Diagnostic::parse_error(
            span.clone(),

            None,
        ));
    }
}

// Rama para events [...]
Token::Events => {
    // Consumir '['
    expect_token!(tokens, Token::LBracket, diags, span, "'[' despues de 'events'");
    loop {
        match tokens.next() {
            Some((Token::RBracket, _)) => break,
            Some((Token::Ident(name), sp)) => {
                endpoint_emits.push(Spanned::new(name, sp));
                // Consumir coma opcional
                if matches!(tokens.peek(), Some((Token::Comma, _))) {
                    tokens.next();
                }
            }
            _ => {

        }
```

- [ ] **Step 4: Agregar el parser del bloque `event`**

Agregar una funcion `parse_event_block` en el modulo del parser:

```rust
fn parse_event_block(
    tokens: &mut PeekableTokens,
    diags: &mut Vec<Diagnostic>,
    start: Span,
) -> Option<EventDef> {
    // nombre del evento (puede contener punto: user.created)
    let (name, name_span) = match tokens.next() {
        Some((Token::Ident(n), sp)) => (n, sp),
        _ => {
            diags.push(Diagnostic::parse_error(start, "nombre de evento esperado", None));
            return None;
        }
    };
    // Consumir '{'
    expect_token_opt!(tokens, Token::LBrace, diags, name_span.clone(), "'{' esperado");

    let mut payload: Option<Spanned<String>> = None;
    let mut retain_days: Option<u32> = None;

    loop {
        match tokens.next() {
            Some((Token::RBrace, _)) => break,
            Some((Token::Payload, sp)) => {
                if let Some((Token::Ident(t), tsp)) = tokens.next() {
                    payload = Some(Spanned::new(t, tsp));
                }
            }
            Some((Token::Retain, sp)) => {
                // Formato: "30d" como un solo token o numero + 'd'
                if let Some((Token::IntLit(n), _)) = tokens.next() {
                    // Consumir 'd' si esta como token separado
                    if matches!(tokens.peek(), Some((Token::Ident(s), _)) if s == "d") {
                        tokens.next();
                    }
                    retain_days = Some(n.abs() as u32);
                }
            }
            None => break,
            _ => {}
        }
    }

    Some(EventDef {
        name: Spanned::new(name, name_span.clone()),
        payload: payload.unwrap_or_else(|| Spanned::new(String::new(), name_span.clone())),
        retain_days,
        span: start,
    })
}
```

En el loop principal del parser (donde se despacha `Token::Model`, `Token::Endpoint`, etc.), agregar la rama:

```rust
Token::Event => {
    if let Some(ev) = parse_event_block(&mut tokens, &mut diags, span) {

    }
}
```

```rust
//! Generador AsyncAPI 2.6 desde el AST del Anti-DSL.
//!
//! Solo se genera si el schema contiene al menos un evento declarado.

use crate::ast::Schema;

/// Genera la especificacion AsyncAPI 2.6 como string YAML.
///
/// Retorna `None` si el schema no tiene eventos declarados.
pub fn generate_async_api(schema: &Schema) -> Option<String> {
    if schema.events.is_empty() {
        return None;
    }

    let project_name = schema.config.as_ref()
        .and_then(|c| c.project_name.as_deref())
        .unwrap_or("anti-gravital-app");

    let mut out = String::new();
    out.push_str("asyncapi: '2.6.0'\n");
    out.push_str(&format!("info:\n  title: {}\n  version: '0.1.0'\n", project_name));
    out.push_str("channels:\n");

    for ev in &schema.events {
        let channel = ev.name.value.replace('.', "/");
        out.push_str(&format!("  {}:\n", channel));
        out.push_str("    subscribe:\n");
        out.push_str("      message:\n");
        out.push_str(&format!("        payload:\n          $ref: '#/components/schemas/{}'\n",
            ev.payload.value));
        if let Some(days) = ev.retain_days {
            out.push_str(&format!("        bindings:\n          nats:\n            retentionDays: {}\n",
                days));
        }
    }

    out.push_str("components:\n  schemas:\n");
    for ev in &schema.events {
        out.push_str(&format!("    {}:\n      type: object\n", ev.payload.value));
    }

    Some(out)
}
```
