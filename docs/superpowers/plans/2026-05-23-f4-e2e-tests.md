//! Tests de integracion Fase 4.

//! Tests unitarios (uno por modulo) y test cross-module E2E.

// ---------------------------------------------------------------------------
// Helpers
// ---------------------------------------------------------------------------

/// Genera un par de claves Ed25519 PKCS#8 PEM para tests de JWT.
fn generate_ed25519_keypair() -> (String, String) {
    use ed25519_dalek::pkcs8::spki::der::pem::LineEnding;
    use ed25519_dalek::pkcs8::{EncodePrivateKey, EncodePublicKey};
    use ed25519_dalek::SigningKey;
    use rand_core::OsRng;

    let signing_key = SigningKey::generate(&mut OsRng);
    let priv_pem = signing_key
        .to_pkcs8_pem(LineEnding::LF)
        .unwrap()
        .to_string();
    let pub_pem = signing_key
        .verifying_key()
        .to_public_key_pem(LineEnding::LF)
        .unwrap();
    (priv_pem, pub_pem)
}

fn make_auth_config(priv_pem: String, pub_pem: String) -> ag_auth::AuthConfig {
    ag_auth::AuthConfig {
        jwt_private_key_pem: priv_pem,
        jwt_public_key_pem: pub_pem,
        webauthn_rp_id: String::new(),
        webauthn_origin: String::new(),
        oauth_google_client_id: None,
        oauth_google_client_secret: None,
        oauth_github_client_id: None,
        oauth_github_client_secret: None,
    }
}

fn make_jwt_claims(sub: &str, role: &str) -> ag_auth::Claims {
    use std::time::{SystemTime, UNIX_EPOCH};
    let now = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .unwrap()
        .as_secs();
    ag_auth::Claims {
        sub: sub.to_string(),
        exp: now + 3600,
        iat: now,
        jti: uuid_like(),
        role: role.to_string(),
    }
}

fn uuid_like() -> String {
    let mut b = [0u8; 16];
    getrandom::getrandom(&mut b).unwrap();
    format!(
        "{:08x}-{:04x}-{:04x}-{:04x}-{:012x}",
        u32::from_be_bytes(b[0..4].try_into().unwrap()),
        u16::from_be_bytes(b[4..6].try_into().unwrap()),
        u16::from_be_bytes(b[6..8].try_into().unwrap()),
        u16::from_be_bytes(b[8..10].try_into().unwrap()),
        {
            let mut n = 0u64;
            for i in 10..16 { n = (n << 8) | b[i] as u64; }
            n
        }
    )
}

#[tokio::test]
async fn auth_jwt_sign_verify_roundtrip() {
    let (priv_pem, pub_pem) = generate_ed25519_keypair();
    let config = make_auth_config(priv_pem, pub_pem);
    let auth = ag_auth::AgAuth::new(config, reqwest::Client::new())
        .expect("AgAuth::new debe tener exito");

    let claims = make_jwt_claims("user-integration-1", "admin");
    let token = auth.jwt.sign(&claims).expect("sign debe tener exito");
    let verified = auth.jwt.verify(&token).expect("verify debe tener exito");

    assert_eq!(verified.sub, "user-integration-1");
    assert_eq!(verified.role, "admin");
}

// ag-cache: L1 set, get, invalidate por tag

#[tokio::test]
async fn cache_l1_set_get_tag_invalidation() {
    use ag_cache::L1Cache;

    let cache = L1Cache::new(ag_cache::CacheConfig::default());
    let key = "user:1:profile";
    let payload = b"json-payload".to_vec();
    let tags = vec!["user:1".to_string()];

    cache.set_bytes_tagged(key, payload.clone(), tags, None).await;

    let hit = cache.get_bytes(key).await;
    assert_eq!(hit.as_deref(), Some(payload.as_slice()), "debe haber cache hit");

    cache.invalidate_tag("user:1").await;
    let miss = cache.get_bytes(key).await;
    assert!(miss.is_none(), "despues de invalidar el tag debe haber miss");
}

        .expect("broadcast debe tener exito");

    let event = tokio::time::timeout(
        std::time::Duration::from_secs(1),

    .expect("debe recibir evento");

    assert_eq!(event.subject, "integration.test");
    assert_eq!(event.payload, b"hello");
}

```rust
// ---------------------------------------------------------------------------
// E2E cross-module: auth + cache + realtime + storage
// ---------------------------------------------------------------------------

/// Test de integracion cruzada: los 5 modulos de Fase 4 cooperan correctamente.
///
/// Flujo:
/// 1. ag-observe: init tracing
/// 2. ag-auth: generar par de claves Ed25519, construir AgAuth
/// 3. ag-auth: firmar JWT con claims {sub: "user-42", role: "admin"}
/// 4. ag-auth: verificar JWT, extraer claims
/// 5. ag-cache: construir L1Cache
/// 6. ag-cache: set("user:42:profile", json_payload, tags=["user:42"])
/// 7. ag-realtime: construir AgRealtime InProcess
/// 8. ag-realtime: subscribe("user.profile.updated")
/// 9. ag-realtime: broadcast_json("user.profile.updated", payload)
/// 10. ag-realtime: verificar que el suscriptor recibe el evento
/// 11. ag-cache: get("user:42:profile") -> hit
/// 12. ag-cache: invalidate_tag("user:42") -> elimina entrada
/// 13. ag-cache: get("user:42:profile") -> None
/// 14. ag-storage: put("avatars/user-42.png", bytes)
/// 15. ag-storage: get("avatars/user-42.png") -> bytes identicos
#[tokio::test]
async fn e2e_authenticated_cached_event() {
    use ag_cache::L1Cache;
    use ag_realtime::{AgRealtime, RealtimeConfig};
    use bytes::Bytes;

    // Paso 1: ag-observe init (puede fallar si ya hay subscriber — no critico)
    let _ = ag_observe::observe_init(&ag_observe::ObserveConfig {
        service_name: "e2e-test".into(),
        log_format: ag_observe::LogFormat::Text,
        otlp_endpoint: None,
        enable_console: false,
        enable_metrics: false,
        metrics_port: 9091,
    });

    // Paso 2: ag-auth — par de claves y AgAuth
    let (priv_pem, pub_pem) = generate_ed25519_keypair();
    let config = make_auth_config(priv_pem, pub_pem);
    let auth = ag_auth::AgAuth::new(config, reqwest::Client::new())
        .expect("AgAuth::new debe tener exito");

    // Paso 3: firmar JWT
    let claims = make_jwt_claims("user-42", "admin");
    let token = auth.jwt.sign(&claims).expect("sign debe tener exito");

    // Paso 4: verificar JWT
    let verified = auth.jwt.verify(&token).expect("verify debe tener exito");
    assert_eq!(verified.sub, "user-42", "sub del token debe ser user-42");
    assert_eq!(verified.role, "admin");

    // Paso 5: ag-cache — L1Cache
    let cache = L1Cache::new(ag_cache::CacheConfig::default());

    // Paso 6: set con tag
    let profile_json = serde_json::json!({
        "id": "user-42",
        "name": "Test User",
        "email": "user42@example.com"
    });
    let profile_bytes = serde_json::to_vec(&profile_json).unwrap();
    cache.set_bytes_tagged(
        "user:42:profile",
        profile_bytes.clone(),
        vec!["user:42".to_string()],
        None,
    )
    .await;

    // Paso 7: ag-realtime — AgRealtime InProcess
    let rt = AgRealtime::new(RealtimeConfig::default())
        .await
        .expect("AgRealtime::new debe tener exito");
    let bus = rt.bus().expect("InProcess debe exponer bus");

    // Paso 8: subscribe antes del broadcast
    let mut rx = bus.subscribe();

    // Paso 9: broadcast_json
    #[derive(serde::Serialize)]
    struct ProfileUpdateEvent {
        user_id: &'static str,
        updated_fields: Vec<&'static str>,
    }
    let event_payload = ProfileUpdateEvent {
        user_id: "user-42",
        updated_fields: vec!["name", "email"],
    };
    rt.broadcast_json("user.profile.updated", &event_payload)
        .expect("broadcast_json debe tener exito");

    // Paso 10: verificar que el suscriptor recibe el evento
    let received = tokio::time::timeout(
        std::time::Duration::from_secs(1),
        rx.recv(),
    )
    .await
    .expect("timeout — evento no recibido en 1 segundo")
    .expect("error al recibir evento");

    assert_eq!(
        received.subject, "user.profile.updated",
        "subject del evento incorrecto"
    );
    let received_val: serde_json::Value =
        serde_json::from_slice(&received.payload).expect("payload debe ser JSON valido");
    assert_eq!(received_val["user_id"], "user-42");

    // Paso 11: cache hit
    let cached = cache.get_bytes("user:42:profile").await;
    assert!(cached.is_some(), "debe haber cache hit (L1 calido)");
    assert_eq!(
        cached.as_deref(),
        Some(profile_bytes.as_slice()),
        "payload cacheado debe coincidir"
    );

    // Paso 12: invalidar tag
    cache.invalidate_tag("user:42").await;

    // Paso 13: cache miss post-invalidacion
    let post_invalidation = cache.get_bytes("user:42:profile").await;
    assert!(
        post_invalidation.is_none(),
        "despues de invalidar el tag, la entrada debe haber sido eliminada"
    );

    // Paso 14: ag-storage — put
    let dir = tempfile::tempdir().expect("tempdir");
    let storage_config = ag_storage::StorageConfig {
        root_path: dir.path().to_path_buf(),
        ..ag_storage::StorageConfig::default()
    };
    let storage = ag_storage::AgStorage::new(storage_config)
        .await
        .expect("AgStorage::new debe tener exito");

    let avatar_data = Bytes::from(vec![0x89u8, 0x50, 0x4E, 0x47, 0x0D, 0x0A]); // PNG magic bytes
    storage
        .put("avatars/user-42.png", avatar_data.clone())
        .await
        .expect("put avatar debe tener exito");

    // Paso 15: ag-storage — get
    let retrieved = storage
        .get("avatars/user-42.png")
        .await
        .expect("get avatar debe tener exito");

    assert_eq!(
        retrieved, avatar_data,
        "bytes recuperados deben coincidir con los almacenados"
    );

    // Verificacion final: api key de auth tambien funciona en este contexto
    let (raw_key, key_hash) = auth.create_api_key("e2e");
    assert!(auth.verify_api_key(&raw_key, &key_hash), "api key debe verificar en contexto E2E");
}
```
